    47→// ─── Reveal Animation Wrapper ───
    48→function Reveal({ children, className = '', delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
    49→  const ref = useRef(null)
    50→  const isInView = useInView(ref, { once: true, margin: '-50px' })
    51→  return (
    52→    <motion.div
    53→      ref={ref}
    54→      initial={{ opacity: 0, y: 40 }}
    55→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
    56→      transition={{ duration: 0.6, delay, ease: 'easeOut' }}
    57→      className={className}
    58→    >
    59→      {children}
    60→    </motion.div>
    61→  )
    62→}
