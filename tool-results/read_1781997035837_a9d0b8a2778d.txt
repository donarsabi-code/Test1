    49→// ─── Reveal Animation Wrapper ───
    50→function Reveal({ children, className = '', delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
    51→  const ref = useRef(null)
    52→  const isInView = useInView(ref, { once: true, margin: '-50px' })
    53→  return (
    54→    <motion.div
    55→      ref={ref}
    56→      initial={{ opacity: 0, y: 40 }}
    57→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
    58→      transition={{ duration: 0.6, delay, ease: 'easeOut' }}
    59→      className={className}
    60→    >
    61→      {children}
    62→    </motion.div>
    63→  )
    64→}

   292→// ─── Generate PDF ───
   293→async function generateStudentPDF(studentData: Record<string, unknown>) {
   294→  const { default: jsPDF } = await import('jspdf')
   295→  const doc = new jsPDF()
   296→  const s = studentData
   297→
   298→  // Header
   299→  doc.setFillColor(37, 99, 235)
   300→  doc.rect(0, 0, 210, 40, 'F')
   301→  doc.setTextColor(255, 255, 255)
   302→  doc.setFontSize(20)
   303→  doc.text('ESTAM', 15, 20)
   304→  doc.setFontSize(10)
   305→  doc.text('Ecole Superieure des Technologies Avancees et de Management', 15, 30)
   306→
   307→  // Student Info
   308→  doc.setTextColor(0, 0, 0)
   309→  doc.setFontSize(16)
   310→  doc.text('Fiche Etudiant', 15, 55)
   311→
   312→  doc.setFontSize(11)
   313→  const fields = [
   314→    ['Identifiant', String(s.studentId || '-')],
   315→    ['Nom complet', `${String(s.firstName || '')} ${String(s.lastName || '')}`],
   316→    ['Email', String(s.email || '-')],
   317→    ['Telephone', String(s.phone || '-')],
   318→    ['Ville', String(s.city || '-')],
   319→    ['Nationalite', String(s.nationality || '-')],
   320→    ['Genre', String(s.genre === 'M' ? 'Masculin' : s.genre === 'F' ? 'Feminin' : '-')],
   321→    ['Date de naissance', String(s.dateOfBirth || '-')],
   322→    ['Adresse', String(s.address || '-')],
   323→    ['Filiere', String(s.filiere || '-')],
   324→    ['Domaine', String(s.filiereCategory || '-')],
   325→    ['Niveau', String(s.niveau || '-')],
   326→    ['Annee scolaire', String(s.anneeScolaire || '-')],
   327→    ['Statut', String(s.status || '-')],
   328→    ['Verifie', String(s.verified === true ? 'Oui' : 'Non')],
   329→  ]
   330→
   331→  let y = 65
   332→  for (const [label, value] of fields) {
   333→    doc.setFont(undefined, 'bold')
   334→    doc.setTextColor(80, 80, 80)
   335→    doc.text(`${label} :`, 15, y)
   336→    doc.setFont(undefined, 'normal')
   337→    doc.setTextColor(0, 0, 0)
   338→    doc.text(value, 75, y)
   339→    y += 10
   340→  }
   341→
   342→  // Grades
   343→  const grades = (s.grades as Record<string, string>[]) || []
   344→  if (grades.length > 0) {
   345→    y += 5
   346→    doc.setFontSize(14)
   347→    doc.setTextColor(37, 99, 235)
   348→    doc.text('Notes', 15, y)
   349→    y += 8
   350→    doc.setFontSize(9)
   351→    doc.setTextColor(100, 100, 100)
   352→    doc.text('Matiere', 15, y)
   353→    doc.text('Type', 80, y)
   354→    doc.text('Note', 120, y)
   355→    doc.text('Semestre', 150, y)
   356→    y += 2
   357→    doc.setDrawColor(200, 200, 200)
   358→    doc.line(15, y, 195, y)
   359→    y += 6
   360→    doc.setTextColor(0, 0, 0)
   361→    doc.setFontSize(10)
   362→    for (const g of grades) {
   363→      if (y > 270) { doc.addPage(); y = 20 }
   364→      doc.text(String(g.matiere || ''), 15, y)
   365→      doc.text(String(g.type || ''), 80, y)
   366→      doc.text(String(g.note || ''), 120, y)
   367→      doc.text(String(g.semestre || ''), 150, y)
   368→      y += 8
   369→    }
   370→  }
   371→
   372→  // Payments
   373→  const payments = (s.payments as Record<string, string>[]) || []
   374→  if (payments.length > 0) {
   375→    y += 10
   376→    if (y > 250) { doc.addPage(); y = 20 }
   377→    doc.setFontSize(14)
   378→    doc.setTextColor(37, 99, 235)
   379→    doc.text('Paiements', 15, y)
   380→    y += 8
   381→    doc.setFontSize(9)
   382→    doc.setTextColor(100, 100, 100)
   383→    doc.text('Mois', 15, y)
   384→    doc.text('Montant', 80, y)
   385→    doc.text('Statut', 120, y)
   386→    doc.text('Date', 160, y)
   387→    y += 2
   388→    doc.line(15, y, 195, y)
   389→    y += 6
   390→    doc.setTextColor(0, 0, 0)
   391→    doc.setFontSize(10)
   392→    for (const p of payments) {
   393→      if (y > 270) { doc.addPage(); y = 20 }
   394→      doc.text(String(p.mois || ''), 15, y)
   395→      doc.text(`${Number(p.montant || 0).toLocaleString()} FCFA`, 80, y)
   396→      doc.text(String(p.statut === 'paye' ? 'Paye' : 'Impaye'), 120, y)
   397→      doc.text(String(p.datePaiement || '-'), 160, y)
   398→      y += 8
   399→    }
   400→  }
   401→
   402→  // Footer
   403→  const pageCount = doc.getNumberOfPages()
   404→  for (let i = 1; i <= pageCount; i++) {
   405→    doc.setPage(i)
   406→    doc.setFontSize(8)
   407→    doc.setTextColor(150, 150, 150)
   408→    doc.text(`ESTAM Congo - Fiche etudiante - Genere le ${new Date().toLocaleDateString('fr-FR')}`, 15, 290)
   409→    doc.text(`Page ${i}/${pageCount}`, 180, 290)
   410→  }
   411→
   412→  doc.save(`ESTAM_${String(s.studentId || 'etudiant')}.pdf`)
   413→}

   708→      if (!res.ok) { setError(data.error || 'Erreur'); return }
